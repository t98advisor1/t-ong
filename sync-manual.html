1 통합 Apps Script 코드 (원클릭 클립보드 복사)

아래 코드는 나이스 생기부 웹앱과 디지털 교무수첩(T-ong)의 출결·칭찬·1인1역·수업관찰·상담기록 10개 탭 동기화 및 관리자 가입 승인을 모두 처리하는 통합 코드입니다. 

/**
 * NEIS-HELP & T-ONG 올인원 10개 탭 통합 동기화 웹 앱 (Google Apps Script)
 * 지원: 2_학생명렬표, 3_창체_행특, 5_상담기록, 6_수업관찰_기록, 7_AI세특_작성, 8_출결_칭찬_기록
 */
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    var action = data.action || "log";
    
    // 1. 교사 승인 요청 기록 (관리자용)
    if (action === "auth_request" || action === "register_user") {
      var authSheet = ss.getSheetByName("0_관리자승인") || ss.insertSheet("0_관리자승인");
      if (authSheet.getLastRow() === 0) {
        authSheet.appendRow(["승인여부", "신청일시", "성명", "소속학교", "구글이메일", "연락처/메모", "권한상태", "만료예정일"]);
      }
      authSheet.appendRow([
        false,
        new Date(),
        data.name || "",
        data.school || "",
        data.email || "",
        data.memo || "웹앱 로그인 신청",
        "대기",
        data.expiry || ""
      ]);
      return ContentService.createTextOutput(JSON.stringify({ok: true, result: "success", msg: "가입신청 기록완료"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // 2. 디지털 교무수첩(T-ong) 출결 · 칭찬 · 1인1역 수신 -> [8_출결_칭찬_기록]
    if (action === "attendance" || action === "praise" || action === "duty" || action === "tong_record") {
      var attSheet = ss.getSheetByName("8_출결_칭찬_기록");
      if (attSheet) {
        attSheet.appendRow([
          data.date || new Date().toISOString().slice(0, 10),
          data.studentId || "",
          data.studentName || "",
          data.category || (action === "praise" ? "칭찬" : action === "duty" ? "1인1역" : "출결"),
          data.detail || data.status || "",
          data.memo || "",
          "자동반영됨"
        ]);
      }
      return ContentService.createTextOutput(JSON.stringify({ok: true, result: "success", msg: "8_출결_칭찬_기록 동기화완료"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // 3. 디지털 교무수첩(T-ong) 수업관찰 · 세특 메모 수신 -> [6_수업관찰_기록]
    if (action === "observation" || action === "setuk_memo") {
      var obsSheet = ss.getSheetByName("6_수업관찰_기록");
      if (obsSheet) {
        obsSheet.appendRow([
          data.date || new Date().toISOString().slice(0, 10),
          data.subject || data.classCode || "교과수업",
          data.studentId || "",
          data.studentName || "",
          data.unit || "수업관찰",
          data.content || data.memo || "",
          "교무수첩(T-ong)"
        ]);
      }
      return ContentService.createTextOutput(JSON.stringify({ok: true, result: "success", msg: "6_수업관찰_기록 동기화완료"}))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 4. 상담 기록 수신 -> [5_상담기록]
    if (action === "counsel") {
      var cSheet = ss.getSheetByName("5_상담기록");
      if (cSheet) {
        cSheet.appendRow([
          data.date || new Date().toISOString().slice(0, 10),
          data.studentId || "",
          data.studentName || "",
          data.type || "진학·학습",
          data.summary || "",
          data.followUp || ""
        ]);
      }
      return ContentService.createTextOutput(JSON.stringify({ok: true, result: "success", msg: "5_상담기록 동기화완료"}))
        .setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ok: true, result: "received"}))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ok: false, result: "error", error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({ok: true, status: "online", service: "NEIS-HELP & T-ONG ALL-IN-ONE GAS"}))
    .setMimeType(ContentService.MimeType.JSON);
}
